
### Валидация форм

```typescript
const validate = () => {
  const newErrors: Record<string, string> = {};

  if (!formData.title.trim()) {
    newErrors.title = 'Название обязательно';
  } else if (formData.title.length < 3) {
    newErrors.title = 'Минимум 3 символа';
  }

  setErrors(newErrors);
  return Object.keys(newErrors).length === 0;
};
```
